```python
def sec(p, k):                       # parser minimo de .secrets.env (so as chaves pedidas)
    o = {}
    if not os.path.exists(p): return o
    for l in open(p, encoding="utf-8", errors="replace"):
        l = l.strip()
        if l and not l.startswith("#") and "=" in l:
            a, b = l.split("=", 1)
            if a.strip() in k: o[a.strip()] = b.strip().strip('"').strip("'")
    return o

S = sec(os.path.expanduser("~/.claude/.secrets.env"), {"TELEGRAM_BOT_TOKEN", "TELEGRAM_CHAT_ID"})

def telegram(msg):                   # urllib.request, stdlib — nao depende de `requests`
    r = urllib.request.Request(
        f"https://api.telegram.org/bot{S['TELEGRAM_BOT_TOKEN']}/sendMessage",
        data=json.dumps({"chat_id": S["TELEGRAM_CHAT_ID"], "text": msg,
                         "parse_mode": "HTML"}).encode(), method="POST")
    r.add_header("Content-Type", "application/json")
    with urllib.request.urlopen(r, timeout=25) as x:
        return json.loads(x.read().decode()).get("ok", False)
```
