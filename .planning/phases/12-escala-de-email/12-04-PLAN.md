```python
FREE_WEBMAIL = {

def email_ok(email, seen_emails, seen_domains):
    if not email or '@' not in email:        return False, 'sem-email'
    email = email.strip().lower()
    if '%' in email:                         return False, 'email-malformado'
    dom = email.split('@', 1)[1]
    if email in seen_emails:                 return False, 'dup-email'
    if dom in seen_domains:                  return False, 'dup-dominio'
    if dom in FREE_WEBMAIL:                  return False, 'webmail-nao-corp'   # <-- o portão
    if any(dom.endswith(t) for t in FOREIGN_TLD): return False, 'tld-estrangeiro'
    return True, 'ok'
```
